smart_campus_system.py 
import os
import csv
import json
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt


class StudentProjectDirectoryError(Exception):
    """User-defined exception for invalid student project directories."""


class SmartCampusInformationSystem:
    def __init__(self, record_file="academic_records.json", csv_file="performance.csv"):
        self.record_file = record_file
        self.csv_file = csv_file
        self.students = []

    # Q1. Student Registration and Grade Evaluation
    def evaluate_grade_and_remark(self, score):
        if score >= 90:
            return "A", "Excellent"
        if score >= 75:
            return "B", "Very Good"
        if score >= 60:
            return "C", "Good"
        if score >= 40:
            return "D", "Average"
        return "F", "Needs Improvement"

    def register_student(self, student_id, name, age, department, score):
        grade, remark = self.evaluate_grade_and_remark(score)
        student = {
            "student_id": student_id,
            "name": name,
            "age": age,
            "department": department,
            "score": score,
            "grade": grade,
            "remark": remark,
            "courses": [],
            "fees": 0
        }
        self.students.append(student)
        return student

    # Q2. Course Enrollment Management System
    def add_courses_with_loop(self, student_id, max_courses=5):
        student = self.find_student(student_id)
        if student is None:
            print("Student not found.")
            return

        count = 0
        while True:
            if count == max_courses:
                print("Maximum course limit reached.")
                break

            course_name = input("Enter course name or STOP to finish: ")
            if course_name.upper() == "STOP":
                break

            try:
                credits = int(input("Enter credit value: "))
            except ValueError:
                print("Invalid credit input. Skipping this course.")
                continue

            if credits <= 0:
                print("Credits must be positive. Skipping this course.")
                continue

            student["courses"].append({"course_name": course_name, "credits": credits})
            count += 1

    # Q3. Student Record Data Management using Data Structures
    def show_data_structure_report(self):
        student_names = [student["name"] for student in self.students]
        student_dictionary = {
            student["student_id"]: {
                "name": student["name"],
                "age": student["age"],
                "grade": student["grade"]
            }
            for student in self.students
        }

        coding_event = {"SC101", "SC102", "SC104"}
        sports_event = {"SC102", "SC103", "SC104"}
        common_participants = coding_event.intersection(sports_event)

        print("Student names list:", student_names)
        print("Student record dictionary:", student_dictionary)
        print("Common event participants:", common_participants)

    # Q4. Sorting and Searching of Student IDs
    def bubble_sort_ids(self, student_ids):
        sorted_ids = student_ids[:]
        for outer in range(len(sorted_ids)):
            for inner in range(0, len(sorted_ids) - outer - 1):
                if sorted_ids[inner] > sorted_ids[inner + 1]:
                    sorted_ids[inner], sorted_ids[inner + 1] = sorted_ids[inner + 1], sorted_ids[inner]
        return sorted_ids

    def linear_search_id(self, student_ids, target_id):
        for index, student_id in enumerate(student_ids):
            if student_id == target_id:
                return index
        return -1

    def binary_search_id(self, sorted_ids, target_id):
        low = 0
        high = len(sorted_ids) - 1
        while low <= high:
            middle = (low + high) // 2
            if sorted_ids[middle] == target_id:
                return middle
            if sorted_ids[middle] < target_id:
                low = middle + 1
            else:
                high = middle - 1
        return -1

    # Q5. Student Fee Calculation using Functions
    def calculate_fee(self, tuition_fee, hostel_fee=0, transportation_fee=0):
        return tuition_fee + hostel_fee + transportation_fee

    def update_fee(self, student_id, hostel_required=False, transport_required=False):
        student = self.find_student(student_id)
        if student is None:
            print("Student not found.")
            return 0

        total_credits = sum(course["credits"] for course in student["courses"])
        tuition_fee = total_credits * 2500
        hostel_fee = 18000 if hostel_required else 0
        transportation_fee = 6000 if transport_required else 0
        student["fees"] = self.calculate_fee(tuition_fee, hostel_fee, transportation_fee)
        return student["fees"]

    # Q6. File Handling for Student Academic Records
    def write_records_to_file(self):
        with open(self.record_file, "w", encoding="utf-8") as file:
            json.dump(self.students, file, indent=4)
        print("Student academic records written to file.")

    def read_records_from_file(self):
        try:
            with open(self.record_file, "r", encoding="utf-8") as file:
                self.students = json.load(file)
            print("Stored records read successfully.")
        except FileNotFoundError:
            print("Academic record file is missing.")

    def generate_simple_report(self):
        if not self.students:
            print("No records available for report generation.")
            return
        average_score = sum(student["score"] for student in self.students) / len(self.students)
        pass_count = sum(1 for student in self.students if student["grade"] != "F")
        print("Academic Report")
        print("Total students:", len(self.students))
        print("Average score:", round(average_score, 2))
        print("Passed students:", pass_count)

    # Q7. Directory Scanning with Exception Handling
    def scan_project_directory(self, directory_path):
        try:
            if not os.path.isdir(directory_path):
                raise StudentProjectDirectoryError("Invalid student project directory.")

            for root, folders, files in os.walk(directory_path):
                level = root.replace(directory_path, "").count(os.sep)
                indent = "  " * level
                print(f"{indent}{os.path.basename(root)}/")
                for file_name in files:
                    print(f"{indent}  {file_name}")

        except StudentProjectDirectoryError as error:
            print("User-defined exception:", error)
        except FileNotFoundError:
            print("Missing file or folder.")
        except PermissionError:
            print("Permission denied while scanning directory.")

    # Q8. Student Performance Analysis using NumPy, Pandas, Matplotlib
    def write_performance_csv(self):
        with open(self.csv_file, "w", newline="", encoding="utf-8") as file:
            writer = csv.writer(file)
            writer.writerow(["student_id", "name", "score", "grade", "fees"])
            for student in self.students:
                writer.writerow([
                    student["student_id"],
                    student["name"],
                    student["score"],
                    student["grade"],
                    student["fees"]
                ])

    def analyze_performance_csv(self):
        try:
            data_frame = pd.read_csv(self.csv_file)
        except FileNotFoundError:
            print("Performance CSV file not found.")
            return

        scores = np.array(data_frame["score"])
        print("NumPy mean:", np.mean(scores))
        print("NumPy maximum:", np.max(scores))
        print("Pandas summary:")
        print(data_frame.describe())
        print("Grade counts:")
        print(data_frame["grade"].value_counts())

        plt.figure(figsize=(8, 5))
        plt.bar(data_frame["name"], data_frame["score"], color="#0f766e")
        plt.title("Student Performance Analysis")
        plt.xlabel("Student Name")
        plt.ylabel("Score")
        plt.ylim(0, 100)
        plt.tight_layout()
        plt.show()

    def find_student(self, student_id):
        for student in self.students:
            if student["student_id"] == student_id:
                return student
        return None


def main():
    campus = SmartCampusInformationSystem()

    print("Q1. Student Registration and Grade Evaluation")
    student_id = input("Enter student ID: ")
    name = input("Enter student name: ")
    age = int(input("Enter student age: "))
    department = input("Enter department: ")
    score = float(input("Enter exam score: "))
    student = campus.register_student(student_id, name, age, department, score)
    print("Grade:", student["grade"])
    print("Performance remark:", student["remark"])

    print("\\nQ2. Course Enrollment Management")
    campus.add_courses_with_loop(student_id, max_courses=5)

    print("\\nQ3. Data Structures Report")
    campus.show_data_structure_report()

    print("\\nQ4. Sorting and Searching Student IDs")
    student_ids = [student["student_id"] for student in campus.students]
    sorted_ids = campus.bubble_sort_ids(student_ids)
    target_id = input("Enter student ID to search: ")
    print("Sorted IDs:", sorted_ids)
    print("Linear search index:", campus.linear_search_id(student_ids, target_id))
    print("Binary search index:", campus.binary_search_id(sorted_ids, target_id))

    print("\\nQ5. Fee Calculation")
    hostel = input("Hostel required? (yes/no): ").lower() == "yes"
    transport = input("Transportation required? (yes/no): ").lower() == "yes"
    print("Total payable fee:", campus.update_fee(student_id, hostel, transport))

    print("\\nQ6. File Handling")
    campus.write_records_to_file()
    campus.read_records_from_file()
    campus.generate_simple_report()

    print("\\nQ7. Directory Scanning")
    directory_path = input("Enter student project directory path: ")
    campus.scan_project_directory(directory_path)

    print("\\nQ8. NumPy, Pandas, Matplotlib Analysis")
    campus.write_performance_csv()
    campus.analyze_performance_csv()


if __name__ == "__main__":
    main()
